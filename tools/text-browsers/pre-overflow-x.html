static int webveld_validate_document(struct wv_parser *p, const struct wv_profile *profile, unsigned int flags, char **errmsg)
{
    if (p == NULL || profile == NULL) {
        return WV_EINVAL;
    }
    for (size_t i = 0; i < p->element_count; i++) {
        if (!wv_profile_permits_element(profile, p->elements[i].tag, p->elements[i].namespace_id, flags)) {
            return wv_report_violation(p, i, WV_ERR_ELEMENT_NOT_PERMITTED, errmsg);
        }
    }
    return WV_OK;
}
